import { prisma } from "./prisma";
import {
  CityCode,
  POICategory,
  POIRecord,
  POIServiceItem,
  WeatherSensitivity,
} from "@/types/travel";

function parseTags(raw: string): string[] {
  try {
    const parsed = JSON.parse(raw);
    return Array.isArray(parsed) ? parsed : [];
  } catch {
    return [];
  }
}

function getDefaultPriceLabel(category: POICategory, estimatedCostVnd: number): string {
  if (estimatedCostVnd === 0) {
    return "Miễn phí vào cửa / tham khảo dịch vụ tại chỗ";
  }
  switch (category) {
    case "DINING":
      return "Suất ăn & menu đồ uống dự kiến / khách";
    case "LODGING":
      return "Chi phí lưu trú dự kiến / đêm";
    case "ADVENTURE":
      return "Giá vé trải nghiệm & dịch vụ dự kiến / khách";
    case "RELAX":
      return "Chi phí thư giãn / menu & dịch vụ dự kiến";
    case "SIGHTSEEING":
    default:
      return "Giá vé vào cổng dự kiến / khách";
  }
}

function buildDefaultServices(
  id: string,
  category: POICategory,
  estimatedCostVnd: number
): POIServiceItem[] {
  if (category === "DINING") {
    return [
      {
        id: `${id}-srv-1`,
        name: "Suất ăn chính / món đặc sản & nước uống",
        priceVnd: estimatedCostVnd,
        type: "MENU",
        includedInEstimate: true,
        note: "Mức giá dự kiến trung bình cho 1 khách",
      },
      {
        id: `${id}-srv-2`,
        name: "Gọi thêm món khai vị / tráng miệng / đồ uống",
        priceVnd: Math.round(estimatedCostVnd * 0.4),
        type: "MENU",
        includedInEstimate: false,
        note: "Chi phí phát sinh thêm nếu gọi ngoài suất cơ bản",
      },
    ];
  }

  if (category === "LODGING") {
    return [
      {
        id: `${id}-srv-1`,
        name: "Phòng lưu trú tiêu chuẩn",
        priceVnd: estimatedCostVnd,
        type: "SERVICE",
        includedInEstimate: true,
        note: "Đã bao gồm tiện ích phòng cơ bản",
      },
      {
        id: `${id}-srv-2`,
        name: "Giặt ủi / đưa đón / phụ thu nhận phòng sớm",
        priceVnd: Math.round(estimatedCostVnd * 0.2),
        type: "SERVICE",
        includedInEstimate: false,
        note: "Dịch vụ tính thêm khi có nhu cầu",
      },
    ];
  }

  return [
    {
      id: `${id}-srv-1`,
      name: estimatedCostVnd > 0 ? "Vé vào cổng khu tham quan" : "Vào cổng tham quan",
      priceVnd: estimatedCostVnd,
      type: "TICKET",
      includedInEstimate: true,
      note:
        estimatedCostVnd > 0
          ? "Giá vé tham khảo cơ bản tại điểm"
          : "Không thu phí vé vào cổng",
    },
    {
      id: `${id}-srv-2`,
      name: "Dịch vụ đi kèm (xe điện, thuê đồ, chụp ảnh, ăn nhẹ)",
      priceVnd: estimatedCostVnd > 0 ? Math.round(estimatedCostVnd * 0.5) : 50000,
      type: "SERVICE",
      includedInEstimate: false,
      note: "Tùy chọn thêm nếu sử dụng dịch vụ tại điểm",
    },
  ];
}

export function mapPoiToDomain(row: {
  id: string;
  name: string;
  city: string;
  description: string;
  lat: number;
  lng: number;
  category: string;
  tags: string;
  estimatedCostVnd: number;
  avgDurationMins: number;
  openTime: string;
  closeTime: string;
  weatherSensitivity: string;
  terrainType: string;
  wheelchairAccessible: boolean;
  vegetarianFriendly: boolean;
  avgRating: number;
  reviewCount: number;
}): POIRecord {
  const city = row.city as CityCode;
  const category = row.category as POICategory;
  const weatherSensitivity = row.weatherSensitivity as WeatherSensitivity;

  return {
    ...row,
    city,
    category,
    weatherSensitivity,
    tags: parseTags(row.tags),
    priceLabel: getDefaultPriceLabel(category, row.estimatedCostVnd),
    services: buildDefaultServices(row.id, category, row.estimatedCostVnd),
  };
}

export const POIRepository = {
  async findAll(cityFilter?: string): Promise<POIRecord[]> {
    const rows = await prisma.pOI.findMany({
      where: cityFilter ? { city: cityFilter } : undefined,
      orderBy: { avgRating: "desc" },
    });
    return rows.map(mapPoiToDomain);
  },

  async findByCities(cities: CityCode[]): Promise<POIRecord[]> {
    const rows = await prisma.pOI.findMany({
      where: { city: { in: cities } },
    });
    return rows.map(mapPoiToDomain);
  },

  async create(data: Omit<POIRecord, "id" | "avgRating" | "reviewCount">) {
    const { priceLabel: _priceLabel, services: _services, ...dbData } = data;
    const row = await prisma.pOI.create({
      data: {
        ...dbData,
        tags: JSON.stringify(dbData.tags),
      },
    });
    return mapPoiToDomain(row);
  },

  async update(id: string, data: Partial<Omit<POIRecord, "id">>) {
    const { priceLabel: _priceLabel, services: _services, ...dbData } = data;
    const row = await prisma.pOI.update({
      where: { id },
      data: {
        ...dbData,
        tags: dbData.tags ? JSON.stringify(dbData.tags) : undefined,
      },
    });
    return mapPoiToDomain(row);
  },

  async delete(id: string) {
    return prisma.pOI.delete({ where: { id } });
  },
};

export const TripRepository = {
  async findByIdWithDetails(id: string) {
    return prisma.trip.findUnique({
      where: { id },
      include: {
        user: { select: { id: true, name: true, email: true } },
        itineraryItems: {
          orderBy: [{ dayNumber: "asc" }, { orderIndex: "asc" }],
          include: { poi: true },
        },
        budgetBreakdowns: true,
        reviews: {
          orderBy: { createdAt: "desc" },
          include: { user: { select: { name: true } }, poi: { select: { name: true } } },
        },
      },
    });
  },

  async findUserTrips(userId: string) {
    return prisma.trip.findMany({
      where: { userId },
      orderBy: { createdAt: "desc" },
      include: {
        budgetBreakdowns: true,
        _count: { select: { itineraryItems: true, reviews: true } },
      },
    });
  },

  async findTopCommunityPlans(limit = 6) {
    return prisma.trip.findMany({
      where: { isPublicCommunity: true },
      orderBy: [{ avgRating: "desc" }, { reviewCount: "desc" }, { cloneCount: "desc" }],
      take: limit,
      include: {
        user: { select: { name: true } },
        itineraryItems: {
          take: 4,
          orderBy: [{ dayNumber: "asc" }, { orderIndex: "asc" }],
          select: { title: true, city: true, jaccardScore: true },
        },
      },
    });
  },
};
